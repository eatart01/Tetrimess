      let c = 0;

      while (c < cols) {

        if (matrix[r][c] === 1) {

          const runStart = c; while (c < cols && matrix[r][c] === 1) c++;

        }

        i = j + 1;

      }

    });

    const compoundBody = Body.create({ parts: finalParts, isStatic, friction: fr, frictionStatic: frS, frictionAir: sharedFrictionAir, restitution: 0, density: sharedDensity });

let prevSnapshot = null;

let currSnapshot = null;

const INTERP_DELAY_MS = 90;

const lerp = (a, b, t) => a + (b - a) * t;

function lerpAngle(a, b, t) {

  let diff = b - a;

  while (diff > Math.PI) diff -= Math.PI * 2;

  while (diff < -Math.PI) diff += Math.PI * 2;

  return a + diff * t;

}

function getInterpolatedBodies() {

  if (!currSnapshot) return [];

  if (!prevSnapshot) return currSnapshot.bodies;

  const renderTime = performance.now() - INTERP_DELAY_MS;

  if (renderTime <= prevSnapshot.time) return prevSnapshot.bodies;

  if (renderTime >= currSnapshot.time) return currSnapshot.bodies;

  const span = currSnapshot.time - prevSnapshot.time;

  const t = span > 0 ? (renderTime - prevSnapshot.time) / span : 1;

  const prevMap = new Map(prevSnapshot.bodies.map(b => [b.id, b]));

  return currSnapshot.bodies.map(currBody => {

    const prevBody = prevMap.get(currBody.id);

    if (!prevBody) return currBody;

    return {

      ...currBody,

      x: lerp(prevBody.x, currBody.x, t),

      y: lerp(prevBody.y, currBody.y, t),

      angle: lerpAngle(prevBody.angle, currBody.angle, t),

      parts: currBody.parts.map((currPart, idx) => {

        const prevPart = prevBody.parts[idx];

        if (!prevPart) return currPart;

        return { ...currPart, x: lerp(prevPart.x, currPart.x, t), y: lerp(prevPart.y, currPart.y, t) };

      })

    };

  });

}

function drawTileRun(ctx, cx, cy, angle, count, tileSize, vertical) {

  const cos = Math.cos(angle); const sin = Math.sin(angle); const half = tileSize / 2;

  for (let i = 0; i < count; i++) {

    const offset = (i - (count - 1) / 2) * tileSize; const localX = vertical ? 0 : offset; const localY = vertical ? offset : 0;

    const corners = [{ x: -half, y: -half }, { x: half, y: -half }, { x: half, y: half }, { x: -half, y: half }];

    ctx.beginPath();

    corners.forEach((corner, idx) => {

      const wx = cx + (localX + corner.x) * cos - (localY + corner.y) * sin; const wy = cy + (localX + corner.x) * sin + (localY + corner.y) * cos;

      if (idx === 0) ctx.moveTo(wx, wy); else ctx.lineTo(wx, wy);

    });

    ctx.closePath(); ctx.fill(); ctx.stroke();

  }

}

function drawMatrixToCanvas(targetCtx, matrix, color, centerX, centerY, tileSize, alpha = 1.0, isCard = false) {

  const rows = matrix.length; const cols = matrix[0].length; const startX = centerX - (cols * tileSize) / 2; const startY = centerY - (rows * tileSize) / 2;

  targetCtx.save(); targetCtx.globalAlpha = alpha; targetCtx.fillStyle = color; targetCtx.strokeStyle = isCard ? '#27272a' : '#18181c'; targetCtx.lineWidth = 1;

  targetCtx.restore();

}
